package com.example.testtimonbot.service;

public class MessageData {

    public MessageData(Long value1, Long value2) {
        this.value1 = value1;
        this.value2 = value2;
    }

    public Long getValue1() {
        return value1;
    }

    public Long getValue2() {
        return value2;
    }

    @Override
    public String toString() {
        return "MessageData{" +
                "value1=" + value1 +
                ", value2=" + value2 +
                '}';
    }
}
